\begin{definition}
$F$ --- поле, $\chr F \neq 2$, $V/ F$ --- конечномерное векторное пространство. Отображение $q: V \longrightarrow F$ называется квадратичной формой, если 
\begin{gather*}
q(av) = a^2 q(v) \\
B(u, v) = \frac 1 2 (q(u + v) - q(u) - q(v))\text{ --- билинейная форма}
\end{gather*}
\end{definition}

Нетрудно заметить, что существуеут взаимнооднозначное соответсвие между квадратичными формами и билинейными симметричными формами.

Пусть $e_1, e_2, \ldots, e_n$ --- базис $V$. Тогда 

$$q(x_1e_1 + x_2 e_2 + \ldots + x_n e_n) = \sum B(e_i, e_j)x_ix_j$$

Матрица $(B(e_i, e_j))$ называется матрицой квадратичной формы $q$ в базисе $\{e_i\}$. Нетрудно видеть, что при замене базиса $A \rightsquigarrow S^T A S$, так что $\det q \in F^* / F^{*2}$ не зависит от выбора базиса.

Квадратичная форма задаёт линейное отображение в двойственное пространство.

$$f: V \rightarrow V^*: v \rightarrow (u \mapsto B(u, v))$$

\begin{proposition}
    $f$ --- изоморфизм тогда и только тогда, когда $B$ невырождена.
\end{proposition}

\begin{definition}
    $q$ называется изотропной, если существует ненулевой вектор $v$ такой, что $q(v) = 0$. В противном случае $q$ называется анизотропной.
\end{definition}

\begin{definition}
    Пусть $(V_1, q_1)$, $(V_2, q_2)$ --- пространства с квадратичными формами. Тогда на $V_1 \oplus V_2$ определена квадратичная форма $q_1 \bot q_2(v_1, v_2) := q_1(v_1) + q_2(v_2)$.
\end{definition}

\begin{definition}
    Аналогично на $V_1 \otimes V_2$ определена квадратчиная форма $q_1 \otimes q_2(v_1 \otimes v_2) = q_1(v_1) q_2(v_2)$.
\end{definition}

\begin{definition}
    Формы $q_1$ и $q_2$ называются изоморфными если существует изоморфизм $f: V_1 \rightarrow V_2$ такой, что $q_1(v_1) = q_2(f(v_1))$.
\end{definition}

\begin{theorem}[Теорема о сокращении]
    Если $f_1 \bot g \simeq f_2 \bot g$, то и $f_1 \simeq f_2$.
\end{theorem}

\begin{theorem}
    Пусть $V_0 \subset V$ --- подпространство, $q_0 = q|V$ --- подформа формы $q$. Если $q_0$ невырождена, то $q \simeq q_0 \bot q_1$.
\end{theorem}
\begin{proof}
    $V_0^\bot = \{v \in V | B_q(v, v_0) = 0 \forall v_0 \in V_0\}$
    Тогда $V_0^\bot \cap V_0 = 0$. Ну действительно, пусть $u$ лежит в пересечении, тогда $B(u, v_0) = 0$ для любого $v_0 \in v$, но тогда образ $u$ в $V_0^*$ равен нулю, значит, т. к. $q_0$ невырождена, $u = 0$.
    Остаётся заметить, что существует точная последовательность
    \begin{equation*}
    \begin{tikzcd}
    0 \rar & V_0^\bot \rar & V \rar & V_0^*
    \end{tikzcd}
    \end{equation*}
    Значит $\dim V = \dim V_0^* + \dim V_0^\bot = \dim V_0 + \dim V_0^\bot$, а значит $V = V_0 \oplus V_0^\bot$.
\end{proof}

\begin{proposition}
    $\{a, -a\} = 0$, $\{a, b\} = \{b, a\}$.
\end{proposition}
\begin{proof}
    $$0 = \{a^{-1}, 1 - a^{-1}\} = \{a^{-1}, \frac{a - 1}{a}\} = -\{a, \frac{a - 1}{a}\} = -(\{a, 1 - a\} - \{a, -a\}) = \{a, -a\}$$
    $$0 = \{ab, -ab\} = \{a, -ab\} + \{b, -ab\} = \{a, -a\} + \{a, b\} + \{b, a\} + \{b, b\}$$
\end{proof}

Тогда можно построить следующую цепочку изоморфизмов: $\langle a_1, a_2, \ldots, a_n \rangle \simeq \langle a_1 x_1^2 + a_2 x_2^2, *, \ldots, a_n \rangle \simeq \ldots \simeq \langle b_1, *, \ldots, * \rangle$.     $n = 2m, l = 2k$

    $\chi(f) + \chi(g) + \left(\frac{m(m - 1)}{2} + \frac{k(k - 1)}{2}\right)\{-1, -1\} = \chi(f \oplus g) + \frac{(m + k)(m + k + 1)}{2}\{-1, 1\}$

    Нетрудно видеть, что

    $\frac{m(m-1)}{2} + \frac{k(k - 1)}{2} + mk = \frac{(m + k)(m + k + 1)}{2}$

Пусть $d \in F^*$ таков, что $\varphi_{F(\sqrt{d})}$ изотропна, тогда $\varphi \simeq \langle x, -xd \rangle \perp \psi$, а значит $c(\psi_F(\sqrt{d}))$ --- кватернионная алгебра над $F(\sqrt{d})$, значит $\psi_F(\sqrt{d})$ изотропна, и $\varphi \simeq \langle x, -xd, y, -yd \rangle \perp \rho$. Получаем, что $c(\varphi) = (d, -xy) + c(\rho) =: (d, -xy) + (a, b)$. Тогда, т. к. $\langle a, b, -ab, -d, xy, -dxy \rangle$ изотропна, найдётся $e \in D(\langle a, b, -ab \rangle) \cap D(\langle d, -xy, dxy\rangle)$. Пусть $e = au^2 + bv^2 - abw^2$, имеем $\llangle a, b \rrangle \simeq \llangle au^2 + bv^2, -ab \rrangle \simeq \llangle e, ab(au^2 + bv^2) \rrangle$, а значит $\langle x, -xd, y, -yd \rangle$ и $\rho$ делятся на $\llangle e \rrangle$.
\end{proof}

\begin{theorem}[Wordsworth, 1975]
    $\dim \varphi_1 = \dim \varphi_2 = 4$, $\disc(\varphi_1) = \disc(\varphi_2) = d$, и $(\varphi_1)_{F(\sqrt{d})} \sim (\varphi_2)_{F(\sqrt{d})}$, тогда $\varphi_1 \sim \varphi_2$.
\end{theorem}
\begin{proof}
    Все свойства в условии не зависят от домножения на константу, так что можем считать
    $\varphi_1 \simeq \langle d, -u_1, -v_1, u_1v_1 \rangle$, $\varphi_2 \simeq \langle d, -u_2, -v_2, u_2v_2\rangle$. Тогда $(\varphi_1)_{F(\sqrt{d})} \simeq \llangle u_1, v_1 \rrangle$, $(\varphi_2)_{F(\sqrt{d})} \simeq \llangle u_2, v_2 \rrangle$, а значит $\llangle u_1, v_1 \rrangle_{F(\sqrt{d})} \simeq \llangle u_2, v_2 \rrangle_{F(\sqrt{d})}$, в частности $\langle u_1, v_1, -u_1v_1, -u_2, -v_2, u_2v_2 \rangle$ изотропна, получаем, что у $\langle u_1, v_1, -u_1 v_1 \rangle$ и $\langle u_2, v_2, -u_2 v_2 \rangle$ есть общее значение, значит $\varphi_1 \simeq \langle d, -a, -b_1, ab_1 \rangle$, $\varphi_2 \simeq \langle d, -a, -b_2, ab_2 \rangle$. Опять же $\llangle a, b_1 \rrangle_{F(\sqrt{d})} \simeq \llangle a, b_2 \rrangle_{F(\sqrt{d})}$, т. е. $\llangle a, b_1b_2 \rrangle \simeq \llangle d, u \rrangle$, $d = ax_1^2 + b_1 b_2 x_2^2 - ab_1 b_2 x_3^2$. Утверждается, что $\varphi_2 \simeq a(ax_1^2 - d) \varphi_1$(упражнение).
\end{proof}

\begin{lemma}
    $\dim \psi \leq 10$, $\psi \in I^2(F)$, $c(\psi) = 0$, тогда $\psi \in I^3$.
\end{lemma}
\begin{proof}
    Случаи $\dim \psi = 4, 6, 8$ были. Пусть $\dim \psi = 10$. $\psi \simeq \langle 1, -a, -b, abd \rangle \perp \tilde{\psi}$, аналогично несуществования анизотропной, $\psi \simeq \langle 1, -a, -b, abd, x, -xd \rangle \perp \tau$, значит первая форма изотропна, значит и $\psi$ изотропна. 
\end{proof}

\begin{theorem}
    $(a_1, b_1) \otimes (c_1, d_1) \simeq (a_2, b_2) \otimes (c_2, d_2)$, $\varphi_i = \langle a_i, b_i, -a_ib_i, -c_i, -d_i, c_i d_i \rangle$, для $i = 1, 2$.
    \begin{enum}{\roman*)}
        \item $\varphi_1 \equiv \varphi_2 \mod I^3$.
        \item $\varphi_1 \sim \varphi_2$.
    \end{enum}
\end{theorem}
\begin{proof}
    Если верно первое утверждение, можно взять $\alpha$, т. ч. $\varphi_1 - \alpha \varphi_2 \in I^3$, $\dim(\varphi_1 \perp -\alpha \varphi_2) = 8$, а значит $\varphi_1 - \alpha \varphi_2 \simeq \pi$, т. е. $\varphi_1 \perp -\pi \simeq \alpha \varphi_2 \perp 4H$. Тогда $\varphi$ и $\pi$ имеют общую подформу размерности $4$. Пусть $\varphi \simeq \tau \perp \rho$, $\pi \simeq \tau \perp \tau_1$. Пусть $d = \det \tau$, тогда $\pi_F(\sqrt{d}) \simeq \tau_F(\sqrt{d}) \perp {\tau_1}_F(\sqrt{d})$. Тогда $0 = c(\pi_F(\sqrt{d})) = c(\tau_F(\sqrt{d})) + c({\tau_1}_F(\sqrt{d}))$, значит $\tau_F(\sqrt{d}) \sim {\tau_1}_F(\sqrt{d})$, значит $\tau \sim \tau_1$, т. е. $\pi \simeq \tau \perp -u\tau$. $\alpha \varphi_2 \simeq \rho \perp u\tau$, или же $\alpha u \varphi_2 \simeq u\rho \perp \tau$, а значит $\varphi_1 - \alpha u \varphi_2 = \rho - u\rho = 0$, по теореме Арасона--Пфистера.

    Докажем первое утверждение. $\dim (\varphi_1 \perp -\alpha\varphi_2)_{an} \leq 10$ и $c(\varphi_1 -\alpha\varphi_2) = 0$, так что $\varphi_1 -\alpha\varphi_2 \in I^3$.
\end{proof}

\begin{definition}
    Формы
    $$\llangle a_1, a_2, \ldots, a_m \rrangle \simeq \llangle b_1, b_2, \ldots, n_n$$
    называются просто $P$-эквивалентными, если $a_k = b_k$ для всех, кроме $i \neq j$, причём $\llangle a_i, a_j \rrangle \simeq \llangle b_i, b_j \rrangle$.
\end{definition}

\begin{definition}
    Формы
    $$\llangle a_1, a_2, \ldots, a_m \rrangle \approx \llangle b_1, b_2, \ldots, n_n$$
    называются $P$-эквивалентными, если они встраиваются в цепочку где соседние формы просто $P$-эквивалентны. 
\end{definition}

\begin{theorem}
    $\varphi = \llangle a_1, \ldots, a_n \rrangle$, $b \in D(\varphi')$, где $\varphi = \langle 1 \rangle \perp \varphi'$. Тогда $\varphi \approx \psi = \llangle -b, \ldots \rrangle$.
\end{theorem}
\begin{proof}
    Докажем по индукции. Случай $n = 1$ очевиден. Пусть теперь, $\tau = \llangle a_1, a_3, \ldots, a_{n - 1} \rrangle$, тогда $\varphi = \tau \perp -a\tau$. $b = x - a_n y$, где $x \in D(\tau') \cup \{0\}$, $y \in D(\tau) \cup \{0\}$.
    \begin{lemma}
        $y \neq 0$, $\llangle a_1, \ldots, a_n \rrangle \approx \llangle a_1, \ldots, a_n y \rrangle$.
    \end{lemma}
    \begin{proof}
        $y = t^2 + y_0$, где $y_0 \in D(\tau')$.
        $$\varphi \approx \llangle -y_0, \ldots, a_n \rrangle,$$
        по индукции, но $\llangle -y_0, a_n \rrangle = \llangle -y_0, a_n y \rrangle$, так что 
        $$\llangle -y_0, \ldots, a_n \rrangle \approx \llangle -y_0, \ldots, a_n y \rrangle \approx \llangle a_1, \ldots, a_n y \rrangle$$
    \end{proof}
    Итого,
    $\tau \approx \llangle -x, d_2, \ldots, d_{n - 1} \rrangle$, $\varphi \approx \llangle -x, d_2, \ldots, d_{n - 1}, a_n \rrangle \approx \llangle -x, d_2, \ldots, d_{n - 1}, a_ny \rrangle \approx \llangle -x + a_n y, \ldots \rrangle$.
\end{proof}

Пусть $L / K$ --- конечное расширение. У нас уже есть $W(F) \xrightarrow{res} W(L)$. Построем отображение $W(L) \rightarrow W(F)$.

Пусть $0 \neq s: L \rightarrow F$ --- $F$-линейное отображение.

Пусть $V \xrightarrow{q} L$ --- квадратичная форма на $V$. Тогда $s \circ q$ --- квадратичная форма, будем обозначать её $s_*(q)$, $s_*$ будем называть трнсфером.

$0 \neq v \in V$, тогда найдётся $u \in V$, т. ч. $0 \neq q(u, v)$. Тогда, для какого-то $x$, $s(x(q(u, v))) \neq 0$, значит $s_*(q)(u, vx) \neq 0$. Так что регулярные переходят в регулярные, а значит мы полуаем отображение между кольцами Витта (гиперболичные очевидно переходят в нулевые).

Пусть $L = F(\sqrt{d})$, $s: L \rightarrow F: 1 \mapsto 0, \sqrt{d} \mapsto 1$. $\langle \alpha + \beta\sqrt{d} \rangle$. $s((\alpha + \beta \sqrt{d})(x + y \sqrt{d})) = \beta(x^2 + dy^2) + 2\alpha xy$, а значит $s_*(\alpha + \beta\sqrt{d}) = \langle \beta, - \beta(\alpha^2 - d \beta^2) \rangle$, при $\beta \neq 0$, иначе форма гиперболична.

\begin{theorem}
    Пусть $L = F(\sqrt{d})$. Тогда треугольник
    \begin{cd}
        W(F) \ar[rr, "res"] & & W(L) \ar[dl, "s_*"] \\
        & W(F) \ar[ul, "\llangle d \rrangle"]
    \end{cd}
    коммутирует и точен.
\end{theorem}
