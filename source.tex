\begin{definition}
$F$ --- поле, $\chr F \neq 2$, $V/ F$ --- конечномерное векторное пространство. Отображение $q: V \longrightarrow F$ называется квадратичной формой, если 
\begin{gather*}
q(av) = a^2 q(v) \\
B(u, v) = \frac 1 2 (q(u + v) - q(u) - q(v))\text{ --- билинейная форма}
\end{gather*}
\end{definition}

Нетрудно заметить, что существуеут взаимнооднозначное соответсвие между квадратичными формами и билинейными симметричными формами.

Пусть $e_1, e_2, \ldots, e_n$ --- базис $V$. Тогда 

$$q(x_1e_1 + x_2 e_2 + \ldots + x_n e_n) = \sum B(e_i, e_j)x_ix_j$$

Матрица $(B(e_i, e_j))$ называется матрицой квадратичной формы $q$ в базисе $\{e_i\}$. Нетрудно видеть, что при замене базиса $A \rightsquigarrow S^T A S$, так что $\det q \in F^* / F^{*2}$ не зависит от выбора базиса.

Квадратичная форма задаёт линейное отображение в двойственное пространство.

$$f: V \rightarrow V^*: v \rightarrow (u \mapsto B(u, v))$$

\begin{proposition}
    $f$ --- изоморфизм тогда и только тогда, когда $B$ невырождена.
\end{proposition}

\begin{definition}
    $q$ называется изотропной, если существует ненулевой вектор $v$ такой, что $q(v) = 0$. В противном случае $q$ называется анизотропной.
\end{definition}

\begin{definition}
    Пусть $(V_1, q_1)$, $(V_2, q_2)$ --- пространства с квадратичными формами. Тогда на $V_1 \oplus V_2$ определена квадратичная форма $q_1 \bot q_2(v_1, v_2) := q_1(v_1) + q_2(v_2)$.
\end{definition}

\begin{definition}
    Аналогично на $V_1 \otimes V_2$ определена квадратчиная форма $q_1 \otimes q_2(v_1 \otimes v_2) = q_1(v_1) q_2(v_2)$.
\end{definition}

\begin{definition}
    Формы $q_1$ и $q_2$ называются изоморфными если существует изоморфизм $f: V_1 \rightarrow V_2$ такой, что $q_1(v_1) = q_2(f(v_1))$.
\end{definition}

\begin{theorem}[Теорема о сокращении]
    Если $f_1 \bot g \simeq f_2 \bot g$, то и $f_1 \simeq f_2$.
\end{theorem}

\begin{theorem}
    Пусть $V_0 \subset V$ --- подпространство, $q_0 = q|V$ --- подформа формы $q$. Если $q_0$ невырождена, то $q \simeq q_0 \bot q_1$.
\end{theorem}
\begin{proof}
    $V_0^\bot = \{v \in V | B_q(v, v_0) = 0 \forall v_0 \in V_0\}$
    Тогда $V_0^\bot \cap V_0 = 0$. Ну действительно, пусть $u$ лежит в пересечении, тогда $B(u, v_0) = 0$ для любого $v_0 \in v$, но тогда образ $u$ в $V_0^*$ равен нулю, значит, т. к. $q_0$ невырождена, $u = 0$.
    Остаётся заметить, что существует точная последовательность
    \begin{equation*}
    \begin{tikzcd}
    0 \rar & V_0^\bot \rar & V \rar & V_0^*
    \end{tikzcd}
    \end{equation*}
    Значит $\dim V = \dim V_0^* + \dim V_0^\bot = \dim V_0 + \dim V_0^\bot$, а значит $V = V_0 \oplus V_0^\bot$.
\end{proof}

\begin{proposition}
    $\{a, -a\} = 0$, $\{a, b\} = \{b, a\}$.
\end{proposition}
\begin{proof}

    $$0 = \{ab, -ab\} = \{a, -ab\} + \{b, -ab\} = \{a, -a\} + \{a, b\} + \{b, a\} + \{b, b\}$$
\end{proof}

$K_2(F) \rightarrow Br F: {a, b} \mapsto (a, b)_{\zeta_n}$

Кроме того, можно определить $K_2(F) / n \simeq {}_nBrF$

\begin{theorem}[Меркурьев--Суслин]
    Этот гомоморфизм на самом деле изоморфизм.
\end{theorem}

Зафиксируем $\zeta_n$, $(a, b)_n := (a, b)_{\zeta_n}$. Пусть $D_1 \simeq (a_1, b_1)_n \otimes (c_1, d_1)_n$, $D_2 \simeq (a_2, b_2) \otimes (c_2, d_2)$, и пусть $D_2 - D_1 = Q$ --- символьная алгебра. Верно ли, что $D_1 \simeq S \otimes R_1$, $D_2 \simeq S \otimes R_2$, где все алгебры символьные? Ответ: нет.

$(a_1, b_1) \otimes (a_2, b_2) \rightsquigarrow \langle a_1, b_1, -a_1b_1, -a_2, -b_2, a_2, b_2 \rangle$ (форма Алберта).

Пусть $I^3(F) = 0$, $D_1, D_2$ --- бикватернионные, и $D_2 - D_1 = Q$. 

$I_2 / I_3 (F) \leftrightarrow {}_2 Br(F)$

$\varphi_2 - \varphi_1 - \pi \in I^3(F) = 0$

$\varphi_2 \perp -\varphi_1 \simeq \pi \perp 4H$

Выделяя по общему значению, получаем, что $\varphi_1 \simeq \tau \perp \rho_1$, $\varphi_2 \simeq \tau \perp \rho_2$, $\dim \tau = 4$.

$\varphi_1 = \langle a_1, a_2, a_3, -a_1a_2a_3 \rangle + \langle -a_1a_2a_3, a_4, u_1, v_1 \rangle \leftrightarrow S + R_1$

$\varphi_2 = \langle a_1, a_2, a_3, -a_1a_2a_3 \rangle + \langle -a_1a_2a_3, a_4, u_2, v_2 \rangle \leftrightarrow S + R_2$

Так что $S \otimes R_1 \simeq D_1$, $S \otimes R_2 \simeq D_2$.

Вернёмся к

\begin{cd}
    0 \rar & W(F) \rar & W(F(x)) \rar & \bigoplus W(F_\pi) \rar & 0
\end{cd}

Докажем сюръективность индукцией по степени $\pi$. Случай $\deg \pi = 1$ очеивден. Далее $\langle \pi h \rangle \mapsto \langle \bar{h} \rangle \oplus (\text{меньшие степени})$, так что по индукции побеждаем.

Инъективность очевидна.

Пусть для $d \geq 0$ $L_d \subset W(F(x))$, где $L_d$ порождена элементами вида $\langle g_1g_2 \ldots g_s \rangle$, где $\deg g_i \leq d$. $\cup L_d = W(F(x))$. Поймём чему равны факторы. Если $\deg \pi = d$, у нас есть гомоморфизм
$$L_d / L_{d-1} \rightarrow W(F_\pi).$$
Хотим понять, что $L_d / L_{d-1} \rightarrow  \bigoplus W(F_\pi)$ --- изоморфизм.

\begin{lemma} \quad
    \begin{enumerate}[nosep, wide, label = \arabic*)]
        \item $L_d / L_{d - 1}$  порождается элементами $\langle \pi g_1 g_2 \ldots g_s \rangle$, где $\deg \pi = d$, $\deg g_i \leq d - 1$.
        \item $\deg \pi = d$, $f$, $g_1, g_2, \ldots, g_s$ таковы, что $\deg f, \deg g_i \leq d - 1$, $f \equiv g_1 g_2 \ldots g_s \mod \pi$. Тогда $\pi f \equiv \pi g_1 g_2 \ldots g_s$.
    \end{enumerate}
\end{lemma}
\begin{proof}
    \begin{enumerate}[nosep, wide, label = \arabic*)]
        \item $\langle \pi_1 \pi_2 \ldots \pi_k g_1 g_2 \ldots g_s \rangle$. Докажем индукцией по $k$.
        $$\pi_1 = \pi_2 + h, \deg h \leq d - 1$$
        $$\langle \pi_2, h \rangle \simeq \langle \pi_1, \pi_1 \pi_2 h \rangle$$
        Домножим обе части на $\pi_1 \pi_3 \ldots \pi_k g_1 g_2 \ldots g_s$.
        $$\langle \pi_1 \pi_2 \ldots \pi_k g_1 g_2 \ldots g_s, \pi_1 \pi_3 \ldots \pi_k g_1 g_2 \ldots g_2h \rangle \simeq \langle \pi_3 \ldots \pi_k g_1 g_2 \ldots g_s, \pi_2 \pi_3 \ldots \pi_k g_1 g_2 \ldots g_s h \rangle$$
        \item Докажем индукцией по $s$, $s = 1$ --- очевидно.
        $$g_1 g_2 = \pi r + h, \deg h \leq d - 1$$
        $$f \equiv h g_3 \ldots g_s \mod \pi$$
        $$\langle \pi r, h \rangle \simeq \langle g_1 g_2, \pi r h g_1 g_2 \rangle$$
        Домножаем на $\pi g_3 \ldots g_s$.
        $$\langle r g_3 \ldots g_s, \pi h g_3 \ldots g_s \rangle \simeq \langle \pi g_1 g_2 \ldots g_s, rh g_1 \ldots g_s \rangle$$
        $$\langle r g_3 \ldots g_s, \pi f \rangle \equiv \langle \pi g_1 g_2 \ldots g_s, rh g_1 \ldots g_s \rangle \mod L_{d - 1}$$
    \end{enumerate}
\end{proof}

Рассмотрим гомоморфизм $\mu_\pi: W(F_\pi) \rightarrow L_d/L_{d - 1}: \langle \bar{f} \rangle \mapsto \langle \pi f \rangle$. Проверим корректность, используя задание $W$ образующими и соотношениями.

\begin{enumerate}[nosep, wide, label = \arabic*)]
    \item $\langle 1 \rangle + \langle -1 \rangle \mapsto 0$
    \item $u = f$, $v = g$, $uv^2 = h$, тогда $\langle u v^2 \rangle - \langle u \rangle \mapsto \langle \pi h \rangle - \langle \pi f \rangle = \langle \pi f g^2 \rangle - \langle \pi f \rangle = 0$.
    \item $\langle u, v \rangle - \langle u + v, uv(u + v) \rangle \mapsto 0$, аналогично.
\end{enumerate}

Осталось проверить, что построенные отображения взаимнообратны. Во одну сторону, по лемме, достаточно проверить, что $\langle \pi g_1 g_2 \ldots g_s \rangle$ переходит в себя, но это очевидно. В обратную сторону очевидно.

На самом деле, последовательность расщепляется. $W(F(X)) \rightarrow W(F): \langle f \rangle \rightarrow \langle l(f) \rangle$ --- старший коэффициент. Итого,
$$W(F(x)) \simeq W(F) \oplus \bigoplus W(F_\pi).$$
\begin{cd}
0 \rar & I^n(F) \rar & I^n(F(x)) \rar & \bigoplus I^{n - 1}(F_\pi) \rar & 0
\end{cd}
