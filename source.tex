\begin{definition}
$F$ --- поле, $\chr F \neq 2$, $V/ F$ --- конечномерное векторное пространство. Отображение $q: V \longrightarrow F$ называется квадратичной формой, если 
\begin{gather*}
q(av) = a^2 q(v) \\
B(u, v) = \frac 1 2 (q(u + v) - q(u) - q(v))\text{ --- билинейная форма}
\end{gather*}
\end{definition}

Нетрудно заметить, что существуеут взаимнооднозначное соответсвие между квадратичными формами и билинейными симметричными формами.

Пусть $e_1, e_2, \ldots, e_n$ --- базис $V$. Тогда 

$$q(x_1e_1 + x_2 e_2 + \ldots + x_n e_n) = \sum B(e_i, e_j)x_ix_j$$

Матрица $(B(e_i, e_j))$ называется матрицой квадратичной формы $q$ в базисе $\{e_i\}$. Нетрудно видеть, что при замене базиса $A \rightsquigarrow S^T A S$, так что $\det q \in F^* / F^{*2}$ не зависит от выбора базиса.

Квадратичная форма задаёт линейное отображение в двойственное пространство.

$$f: V \rightarrow V^*: v \rightarrow (u \mapsto B(u, v))$$

\begin{proposition}
    $f$ --- изоморфизм тогда и только тогда, когда $B$ невырождена.
\end{proposition}

\begin{definition}
    $q$ называется изотропной, если существует ненулевой вектор $v$ такой, что $q(v) = 0$. В противном случае $q$ называется анизотропной.
\end{definition}

\begin{definition}
    Пусть $(V_1, q_1)$, $(V_2, q_2)$ --- пространства с квадратичными формами. Тогда на $V_1 \oplus V_2$ определена квадратичная форма $q_1 \bot q_2(v_1, v_2) := q_1(v_1) + q_2(v_2)$.
\end{definition}

\begin{definition}
    Аналогично на $V_1 \otimes V_2$ определена квадратчиная форма $q_1 \otimes q_2(v_1 \otimes v_2) = q_1(v_1) q_2(v_2)$.
\end{definition}

\begin{definition}
    Формы $q_1$ и $q_2$ называются изоморфными если существует изоморфизм $f: V_1 \rightarrow V_2$ такой, что $q_1(v_1) = q_2(f(v_1))$.
\end{definition}

\begin{theorem}[Теорема о сокращении]
    Если $f_1 \bot g \simeq f_2 \bot g$, то и $f_1 \simeq f_2$.
\end{theorem}

\begin{theorem}
    Пусть $V_0 \subset V$ --- подпространство, $q_0 = q|V$ --- подформа формы $q$. Если $q_0$ невырождена, то $q \simeq q_0 \bot q_1$.
\end{theorem}
\begin{proof}
    $V_0^\bot = \{v \in V | B_q(v, v_0) = 0 \forall v_0 \in V_0\}$
    Тогда $V_0^\bot \cap V_0 = 0$. Ну действительно, пусть $u$ лежит в пересечении, тогда $B(u, v_0) = 0$ для любого $v_0 \in v$, но тогда образ $u$ в $V_0^*$ равен нулю, значит, т. к. $q_0$ невырождена, $u = 0$.
    Остаётся заметить, что существует точная последовательность
    \begin{equation*}
    \begin{tikzcd}
    0 \rar & V_0^\bot \rar & V \rar & V_0^*
    \end{tikzcd}
    \end{equation*}
    Значит $\dim V = \dim V_0^* + \dim V_0^\bot = \dim V_0 + \dim V_0^\bot$, а значит $V = V_0 \oplus V_0^\bot$.
\end{proof}

\begin{definition}
    Кольцо Витта $W(F)$. Пусть $f_1$, $f_2$ --- невырожденные квадратичные формы, если существуют $m, n \geq 0$ такие, что $f_1 \bot mH \simeq f_2 \bot nH$, где $H = \langle 1, -1 \rangle$ --- гиперболическая плоскость.
    $$[f] + [g] = [f \bot g]$$
    $$[f] \otimes [g] = [f \otimes g]$$
\end{definition}

$W(F) \supset I(F)$ --- идеал классов чётномерных форм.

$W(F) \supset I(F) \supset I^2(F) \supset \ldots$

\begin{definition}
    Пусть $\varphi$ --- форма, $D(\varphi) \subset F^*$ --- множество ненулевых значений $\varphi$. $G(\varphi) = \{a \in F^*| \varphi \simeq a \varphi\}$. $G(\varphi)$, очевидно, является подгруппой. Если $1 \in D(\varphi)$, $G(\varphi) \subset D(\varphi)$
\end{definition}

\begin{definition}
    Формой Пфистера называется форма $\llangle a_1, a_2, \ldots, a_n \rrangle := \langle 1, -a_1 \rangle \otimes \langle 1, -a_2 \rangle \otimes \langle 1, -a_3 \rangle \otimes \ldots \otimes \langle 1, -a_n \rangle$.
\end{definition}

\begin{definition}
    $u(F)$ --- максимальная размерность анизотропной квадратчиной формы над $F$.
\end{definition}

$$I \xrightarrow{disc} F^* / F^{*2}$$

\begin{proposition}
    $\ker \disc = I^2(F)$
\end{proposition}
\begin{proof}
    Вложение $I^2 \subset \ker \disc$ очевидно. В обратную сторону $\langle a_1, a_2, \ldots, a_2m \rangle = \langle a_1, a_2, a_3, a_1a_2a_3 \rangle \oplus \langle -a_1a_2a_3, a_4, a_5, \ldots, a_{2m} \rangle$. Первое слагаемое, очевидно лежит в $I^2$, второе имеет размерность на 2 меньше, доказательство завершается индукцией.
\end{proof}

\begin{definition}
    Для любого поля определим $K_0F = \Z$, $K_1 F = F^*$. Для $n \geq 2$ рассмотирим свободную абелеву группу, образующие которой наборы вида $\{a_1, a_2, \ldots, a_n\}$, где $a_i \in F^*$. На этой группе задаются следующие соотношения:
    \begin{enumerate}[wide, label = \arabic*)]
        \item $\{a_1, a_2, \ldots, a_{i-1}, a_ia'_i, a_{i + 1}, \ldots, a_n\} = \{a_1, a_2, \ldots, a_{i - 1}, a_i, a_{i + 1}, \ldots, a_n\} + \{a_1, a_2, \ldots, a_{i - 1}, a'_i, a_{i + 1}, \ldots, a_n\}$.
        \item $\{\ldots, a, \ldots, 1 - a, \ldots\} = 0$
    \end{enumerate}
\end{definition}

\begin{proposition}
    $\{a, -a\} = 0$, $\{a, b\} = \{b, a\}$.
\end{proposition}
\begin{proof}
    $$0 = \{a^{-1}, 1 - a^{-1}\} = \{a^{-1}, \frac{a - 1}{a}\} = -\{a, \frac{a - 1}{a}\} = -(\{a, 1 - a\} - \{a, -a\}) = \{a, -a\}$$
    $$0 = \{ab, -ab\} = \{a, -ab\} + \{b, -ab\} = \{a, -a\} + \{a, b\} + \{b, a\} + \{b, b\}$$
\end{proof}

\begin{definition}
    $k_2(F) := K_2(F) / 2$
\end{definition}

\begin{proposition}
    $I^2(F) / I^3(F) \simeq k_2(F)$
\end{proposition}
\begin{proof}
    $K_2(F) \rightarrow I^2(F) / I^3(F)$ устроен следующим образом: $\{a, b\} \rightarrow \llangle a, b \rrangle (\mod I^3)$.

    $\{a, 1 - a\} \mapsto \llangle a, 1 - a \rrangle = \langle 1, -a, a - 1, a(1 - a)\rangle$

    $\{ab, c\} - \{a, c\} - \{b, c\} \rightarrow \llangle ab, c \rrangle - \llangle a, c \rrangle - \llangle b, c \rrangle = \langle 1, -ab, -c, abc, -1, a, c, -ac, -1, b, c, -bc \rangle = -\langle1, -a, -b, -c, ab, ac, bc, -abc\rangle = - \llangle a, b, c \rrangle$

    Определим отображение в обратную сторону
    
    $f = \langle a_1, a_2, \ldots, a_n \rangle, \disc f = 1$

    $\langle a_1, a_2, \ldots, a_n \rangle \rightsquigarrow \sum \{a_i, a_j\} =: \chi(f)$

    $f \mapsto \chi(f) + \frac{m(m - 1)}{2} \{-1, -1\}$, где $m = \frac{n}{2}$.

    Проверим корректность. Пусть $\langle a_1, a_2, \ldots, a_n\rangle \simeq \langle b_1, b_2, \ldots, b_n \rangle$, тогда $b_1 = a_1x_1^2 + a_2 x_2^2 + \ldots + a_n x_n^2$, причём можно считать, что частичные суммы не равны 0. Тогда можно построить следующую цепочку изоморфизмов: $\langle a_1, a_2, \ldots, a_n \rangle \simeq \langle a_1 x_1^2 + a_2 x_2^2, *, \ldots, a_n \rangle \simeq \ldots \simeq \langle b_1, *, \ldots, * \rangle$. Далее теорема о сокращении. Мы получили, что корректность достаточно проверять для форм отличающихся в двух элементах.

    Пусть $\langle a_1, a_2, a_3, \ldots, a_n\rangle \simeq \langle b_1, b_2, a_3, \ldots, a_n \rangle$. Требуется доказать, что $\{a_1, a_2\} + \{a_1a_2, \prod_{i \geq 3} a_i\} = \{b_1, b_2\} + \{b_1b_2, \prod_{i \geq 3} a_i\}$, но вторые слагаемые не отличаются по модулю квадратов, т. к. детерминанты форм равны по модулю квадратов
    
    Остаётся доказать, что если $\langle a_1, a_2 \rangle \simeq \langle b_1, b_2 \rangle$, то $\{ a_1, a_2 \} \simeq \{ b_1, b_2 \}$.
    
    $b_1 = a_1 x^2 + a_2 y^2$

    $b_2 = a_1a_2(a_1 x^2 + a_2 y^2)$

    $\{a_1x^2 + a_2y^2, a_1a_2(a_1 x^2 + a_2 y^2)\} = \{a_1x^2 + a_2y^2, -a_1a_2\} = \{a_1\left(x^2 + \frac{a_2}{a_1} y^2\right), -a_1a_2\} = \{a_1, -a_1a_2\} + \{x^2 + \frac{a_2}{a_1} y^2, -a_1a_2\} = \{a_1, a_2\} + \{x^2 + \frac{a_2}{a_1} y^2, -\frac{a_2}{a_1}\} = \{a_1, a_2\} + \{x^2 + \frac{a_2}{a_1} y^2, -\frac{a_2}{a_1}y^2\} = \{a_1, a_2\} + \{1 + \frac{a_2}{a_1} \frac{y^2}{x^2}, -\frac{a_2}{a_1}\frac{y^2}{x^2}\} = \{a_1, a_2\}$.

    Нетрудно убедиться, что образ также не зависит от представителя класса.

    $f \simeq \langle a_1, a_2, \ldots, a_n \rangle$

    $g \simeq \langle b_1, b_2, \ldots, b_l \rangle$

    $n = 2m, l = 2k$
    Хотим проверить, что
    
    $\chi(f) + \chi(g) + \left(\frac{m(m - 1)}{2} + \frac{k(k - 1)}{2}\right)\{-1, -1\} = \chi(f \oplus g) + \frac{(m + k)(m + k + 1)}{2}\{-1, 1\}$

    $\chi(f \oplus g) = \chi(f) + \chi(g) + \{\prod a_i, \prod b_j\} = \chi(f) + \chi(g) + mk\{-1, -1\}$

    Нетрудно видеть, что

    $\frac{m(m-1)}{2} + \frac{k(k - 1)}{2} + mk = \frac{(m + k)(m + k + 1)}{2}$

    Нетрудно также понять, что $\chi(\llangle a, b \rrangle) = \chi(c\llangle a, b \rrangle)$.

    $\llangle a, b \rrangle = \langle 1, -a, -b, ab \rangle \mapsto \{-1, -1\} + \{-a, -b\} + \{-a, ab\} + \{-b, ab\} = \{-1, -1\} + \{-a, -1\} + \{-b, a\} = \{a, -1\} + \{-b, a\} = \{a, b\}$
\end{proof}

 \begin{theorem}
    $K_nF / 2 \simeq I^n / I^{n - 1}(F)$.
\end{theorem}
